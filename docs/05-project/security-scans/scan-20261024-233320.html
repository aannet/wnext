docker run --rm \
  --user "$(id -u):$(id -g)" -e HOME=/tmp \
  --read-only --tmpfs /tmp:size=512m \
  --cap-drop ALL --security-opt no-new-privileges \
  --memory 3g \
  -v "$REPO":/src:ro -v "$OUT":/out \
